### 4. Service Worker con Cache Estratégico para Offline

```typescript
// app/sw.ts
import { precacheAndRoute } from 'workbox-precaching';
import { registerRoute } from 'workbox-routing';
import { CacheFirst, NetworkFirst, StaleWhileRevalidate } from 'workbox-strategies';
import { ExpirationPlugin } from 'workbox-expiration';

// Precachear assets críticos
precacheAndRoute(self.__WB_MANIFEST);

// Estrategia: Cursos completos en cache (Cache First)
registerRoute(
  /\/es\/cursos\/.*/,
  new CacheFirst({
    cacheName: 'courses-cache',
    plugins: [
      new ExpirationPlugin({
        maxEntries: 50,
        maxAgeSeconds: 30 * 24 * 60 * 60, // 30 días
      }),
    ],
  }),
  'GET'
);

// Estrategia: Recursos dinámicos (Stale While Revalidate)
registerRoute(
  /\/api\/resources/,
  new StaleWhileRevalidate({
    cacheName: 'resources-cache',
    plugins: [
      new ExpirationPlugin({
        maxEntries: 100,

      }),
    ],
```

### 6. TTS (Text-to-Speech) Multiidioma con Web Speech API + Fallback

```typescript
// hooks/use-text-to-speech.ts
export function useTextToSpeech() {
  const [voices, setVoices] = useState<SpeechSynthesisVoice[]>([]);
  const [speaking, setSpeaking] = useState(false);
  
  useEffect(() => {
    const loadVoices = () => {
      const available = window.speechSynthesis.getVoices();
      setVoices(available);
    };
    
    loadVoices();
    window.speechSynthesis.onvoiceschanged = loadVoices;
  }, []);
  
  const speak = useCallback((text: string, lang: string) => {
    if (!window.speechSynthesis) {
      // Fallback: usar API externa (Google TTS, AWS Polly)
      return playExternalTTS(text, lang);
    }
    
    const utterance = new SpeechSynthesisUtterance(text);
    
    // Seleccionar voz por idioma
    const voice = voices.find(v => v.lang.startsWith(lang)) || 
                  voices.find(v => v.lang.startsWith('es'));
    
    if (voice) {
      utterance.voice = voice;
    }
    
    utterance.lang = lang;
    utterance.rate = 0.9; // Un poco más lento para claridad
    utterance.pitch = 1.0;
    
    utterance.onstart = () => setSpeaking(true);
    utterance.onend = () => setSpeaking(false);
    
    window.speechSynthesis.speak(utterance);
  }, [voices]);
  
  const stop = useCallback(() => {
    window.speechSynthesis.cancel();
    setSpeaking(false);
  }, []);
  
  return { speak, stop, speaking, voices };
}

// Fallback para navegadores sin Web Speech API
async function playExternalTTS(text: string, lang: string) {
  const response = await fetch(`/api/tts`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ text, lang }),
  });
  
  const audioBlob = await response.blob();
  const audio = new Audio(URL.createObjectURL(audioBlob));
  audio.play();
}
```

### 7. Sistema de Progreso Gamificado con XP y Niveles

```typescript
// lib/gamification.ts
interface UserProgress {
  xp: number;
  level: number;
  badges: string[];
  streak: number;
  lastActivity: Date;
}

const LEVEL_THRESHOLDS = [
  0,      // Nivel 1
  100,    // Nivel 2
  250,    // Nivel 3
  500,    // Nivel 4
  1000,   // Nivel 5
  2000,   // Nivel 6
  3500,   // Nivel 7
  5000,   // Nivel 8
  7500,   // Nivel 9
  10000,  // Nivel 10
];

export function calculateLevel(xp: number): number {
  for (let i = LEVEL_THRESHOLDS.length - 1; i >= 0; i--) {
    if (xp >= LEVEL_THRESHOLDS[i]) {
      return i + 1;
    }
  }
  return 1;
}

export function awardXP(
  currentXP: number,
  action: 'lesson_complete' | 'exercise_done' | 'quiz_passed' | 'daily_login'
): number {
  const xpValues = {
    lesson_complete: 20,
    exercise_done: 30,
    quiz_passed: 50,
    daily_login: 10,
  };
  
  return currentXP + xpValues[action];
}

export function checkStreak(lastActivity: Date): number {
  const now = new Date();
  const diffDays = Math.floor((now.getTime() - lastActivity.getTime()) / (1000 * 60 * 60 * 24));
  
  if (diffDays === 0) return 0; // Ya hizo algo hoy
  if (diffDays === 1) return 1; // Racha continúa
  return 0; // Racha rota
}
```

```typescript
// api/finance/education.ts
export async function getPersonalizedFinanceTips(userId: string) {
  const user = await prisma.user.findUnique({
    where: { id: userId },
    include: { bankConnections: true },
  });
  
  if (!user.bankConnections.length) {
    return {
      tips: [
        'Conecta tu cuenta bancaria para recibir consejos personalizados',
        'Aprende a presupuestar con nuestro curso gratuito',
      ],
      courses: ['finanzas-personales-basico'],
    };
  }
  
  // Analizar gastos (con consentimiento)
  const transactions = await fetchTransactions(user.bankConnections[0]);
  
  const categories = categorizeTransactions(transactions);
  
  const tips = [];
  
  if (categories.food > categories.income * 0.3) {
    tips.push('Estás gastando más del 30% en comida. Considera cocinar en casa.');
    tips.push('Curso recomendado: "Ahorro en supermercado"');
  }
  
  if (categories.transport > categories.income * 0.15) {
    tips.push('Tu gasto en transporte es alto. ¿Has considerado abono mensual?');
  }
  
  return { tips, categories, courses: ['ahorro-avanzado'] };
}
```
